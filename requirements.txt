# JWT signing/verification for login sessions (auth.py: `from jose import jwt`).
# Was previously missing from this file entirely — only mentioned in a

# guarantee of getting a working version at all.
python-jose[cryptography]==3.3.0

# Password hashing (auth.py: `from passlib.context import CryptContext`).
# Also previously missing from this file. IMPORTANT: bcrypt is pinned
# below 4.1 deliberately — passlib 1.7.4 (the latest passlib release) is
# incompatible with bcrypt>=4.1 (bcrypt removed the `__about__.__version__`
# attribute passlib's version-detection relies on, and passlib's own
# internal "wrap bug" self-test then raises `ValueError: password cannot
# be longer than 72 bytes` on every single hash call — this fails
# registration and login outright, not just a warning). Confirmed by
# actually hitting this exact failure while testing billing.py against a
# fresh install in this session. Don't bump bcrypt past 4.0.x without
# also confirming passlib has caught up.
passlib==1.7.4
bcrypt==4.0.1

# Encrypting stored teacher/parent-created student credentials at rest
# (dashboard.py). Already pulled in transitively by python-jose[cryptography]
# per auth.py's setup notes, but pinned explicitly here since dashboard.py
# now imports it directly.
cryptography==43.0.3

# PostgreSQL driver for db.py — replaces the old single-file SQLite database.
# SQLite serializes all writes through one file lock and can't be shared
# across multiple app server replicas; see db.py's module docstring for the
# full rationale. Requires a running PostgreSQL server — set DATABASE_URL
# in your .env (see .env.example). psycopg2-binary bundles libpq so there's
# no separate system package to install; switch to the non-binary psycopg2
# build for production if you want to link against your own libpq.
psycopg2-binary==2.9.10

# Optional — used by security.py to share rate-limiting and login-lockout
# state across multiple app replicas via REDIS_URL. Without REDIS_URL set
# (or without this package installed), security.py falls back to
# per-process in-memory state automatically — this is a soft dependency,
# not a hard one. See security.py's module docstring.
redis==5.2.1
